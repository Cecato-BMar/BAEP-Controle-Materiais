    $(document).ready(function() {

        // Fun├º├úo para carregar informa├º├Áes do policial
        function carregarInfoPolicial(policialId) {
            if (!policialId) {
                $('#policial-info').html('');
                return;
            }
            
            $.ajax({
                url: '/policiais/api/policiais/' + policialId + '/',
                type: 'GET',
                dataType: 'json',
                success: function(data) {
                    let html = '<div class="card">';
                    html += '<div class="card-body">';
                    html += '<div class="d-flex align-items-center">';
                    
                    // Foto ou ├¡cone
                    if (data.foto) {
                        html += '<img src="' + data.foto + '" class="rounded-circle me-3" style="width: 60px; height: 60px;">';
                    } else {
                        html += '<div class="bg-secondary text-white rounded-circle d-flex align-items-center justify-content-center me-3" style="width: 60px; height: 60px;"><i class="fas fa-user-shield fa-2x"></i></div>';
                    }
                    
                    // Informa├º├Áes
                    html += '<div>';
                    html += '<h5 class="mb-1">' + data.nome + '</h5>';
                    html += '<p class="mb-1"><strong>RE:</strong> ' + data.re + '</p>';
                    html += '<p class="mb-1"><strong>Posto:</strong> ' + data.posto_display + '</p>';
                    
                    // Situa├º├úo com alerta visual
                    let situacaoClass = 'badge bg-success';
                    let situacaoText = 'Ativo';
                    
                    if (data.situacao !== 'ATIVO') {
                        situacaoClass = 'badge bg-danger';
                        situacaoText = 'Inativo';
                        
                        // Adiciona alerta se o policial n├úo estiver ativo
                        html += '<div class="alert alert-danger mt-2"><i class="fas fa-exclamation-triangle me-2"></i>Este policial n├úo est├í ativo. N├úo ser├í poss├¡vel registrar a devolu├º├úo.</div>';
                    }
                    
                    html += '<p class="mb-0"><strong>Situa├º├úo:</strong> <span class="' + situacaoClass + '">' + situacaoText + '</span></p>';
                    html += '</div>';
                    html += '</div>';
                    html += '</div>';
                    html += '</div>';
                    
                    $('#policial-info').html(html);
                    
                    // Desabilita o bot├úo de envio se o policial n├úo estiver ativo
                    if (data.situacao !== 'ATIVO') {
                        $('button[type="submit"]').prop('disabled', true);
                    } else {
                        $('button[type="submit"]').prop('disabled', false);
                    }
                },
                error: function() {
                    $('#policial-info').html('<div class="alert alert-danger">Erro ao carregar informa├º├Áes do policial.</div>');
                }
            });
        }
        
        // Fun├º├úo para adicionar material para devolu├º├úo
        function adicionarMaterialParaDevolucao(retirada) {
            // Verifica se o material j├í est├í na lista
            const materialExistente = materiaisSelecionados.find(item => 
                item.retirada_id === retirada.id && item.material_id === retirada.material.id
            );
            
            if (materialExistente) {
                alert('Este material j├í foi adicionado ├á lista de devolu├º├úo.');
                return;
            }
            
            // Adiciona o material ├á lista
            materiaisSelecionados.push({
                retirada_id: retirada.id,
                material_id: retirada.material.id,
                nome: retirada.material.nome || retirada.material.tipo_display,
                numero: retirada.material.numero,
                tipo: retirada.material.tipo_display,
                raw_tipo: retirada.material.tipo,
                policial_re: retirada.policial.re,
                policial_nome: retirada.policial.nome,
                quantidade: retirada.quantidade_pendente,
                max: retirada.quantidade_pendente,
                estado: 'BOM', // Estado padr├úo
                manutencao: false, // N├úo enviar para manuten├º├úo por padr├úo
                lote: retirada.lote,
                disparos: 0,
                extravios: 0,
                justificativa_consumo: '',
                boletim_ocorrencia: ''
            });
            
            // Atualiza o campo oculto e a interface
            atualizarCampoOculto();
            atualizarListaMateriais();
        }
        
        // Fun├º├úo para remover material da lista de devolu├º├úo
        function removerMaterialDaDevolucao(index) {
            materiaisSelecionados.splice(index, 1);
            atualizarCampoOculto();
            atualizarListaMateriais();
        }
        
        // Fun├º├úo para atualizar o campo oculto com os materiais selecionados
        function atualizarCampoOculto() {
            $('#id_devolucoes_selecionadas').val(JSON.stringify(materiaisSelecionados));
        }
        
        // Fun├º├úo para atualizar a lista de materiais selecionados na interface
        function atualizarListaMateriais() {

                $('#sem-materiais-msg').show();
                $('#contador-materiais').text('0');
                return;
            }
            
            $('#sem-materiais-msg').hide();
            $('#contador-materiais').text(materiaisSelecionados.length);

            html += '<th>Identifica├º├úo</th><th>Policial</th><th>Quantidade</th><th>Consumo / Estado</th><th>Justificativa / Manuten├º├úo</th><th>A├º├Áes</th>';
            html += '</tr></thead><tbody>';
            
            materiaisSelecionados.forEach(function(item, index) {
                html += '<tr>';
                
                // Coluna 1: Identifica├º├úo
                html += '<td>';
                if (item.raw_tipo === 'MUNICAO') {
                    html += '<span class="badge bg-warning text-dark mb-1"><i class="fas fa-bullseye me-1"></i>' + item.tipo + '</span><br>';
                    html += '<strong>' + item.nome + '</strong>';
                    if (item.lote) {
                        html += '<br><small class="text-muted"><i class="fas fa-barcode me-1"></i>Lote: ' + item.lote.numero_lote + ' (' + item.lote.marca + ')</small>';
                    }

                html += '</td>';

                html += '<td><small>' + item.policial_re + ' - ' + item.policial_nome + '</small></td>';

                if (item.raw_tipo === 'MUNICAO') {
                    const fisicos = item.quantidade - (item.disparos + item.extravios);
                    html += '<div class="mt-2 text-success small"><strong>F├¡sicos: <span id="fisicos-' + index + '">' + fisicos + '</span></strong></div>';
                }
                html += '</td>';

        // Fun├º├úo para adicionar eventos aos controles de materiais
        function adicionarEventosMateriais() {
            // Evento para atualizar a quantidade
            $('.quantidade-material').off('change').change(function() {
                const index = $(this).data('index');
                const novaQuantidade = parseInt($(this).val());
                const item = materiaisSelecionados[index];
                
                if (novaQuantidade > 0 && novaQuantidade <= item.max) {
                    item.quantidade = novaQuantidade;
                    
                    // Se disparos + extravios > quantidade, reseta
                    if (item.disparos + item.extravios > novaQuantidade) {
                        item.disparos = 0;
                        item.extravios = 0;
                        $(`.disparos-municao[data-index="${index}"]`).val(0);
                        $(`.extravios-municao[data-index="${index}"]`).val(0);
                        $(`.consumo-detalhes-${index}`).hide();
                        $(`#consumo-placeholder-${index}`).removeClass('d-none');
                        item.justificativa_consumo = '';
                        item.boletim_ocorrencia = '';
                    }
                    
                    const fisicos = item.quantidade - (item.disparos + item.extravios);
                    $('#fisicos-' + index).text(fisicos);
                    
                    atualizarCampoOculto();
                } else {
                    $(this).val(item.quantidade);
                    alert('Quantidade inv├ílida. Deve ser entre 1 e ' + item.max);
                }
            });
            
            // Evento para disparos de muni├º├úo
            $('.disparos-municao').off('change keyup').on('change keyup', function() {
                const index = $(this).data('index');
                let disparos = parseInt($(this).val()) || 0;
                const item = materiaisSelecionados[index];
                
                if (disparos < 0) {
                    disparos = 0;
                    $(this).val(0);
                }
                
                if (disparos + item.extravios > item.quantidade) {
                    alert('A soma de disparos e extravios n├úo pode exceder a quantidade total de muni├º├Áes prestada de contas (' + item.quantidade + ').');
                    disparos = item.quantidade - item.extravios;
                    $(this).val(disparos);
                }
                
                item.disparos = disparos;
                
                const fisicos = item.quantidade - (item.disparos + item.extravios);
                $('#fisicos-' + index).text(fisicos);
                
                if (item.disparos > 0 || item.extravios > 0) {
                    $('.consumo-detalhes-' + index).show();
                    $('#consumo-placeholder-' + index).addClass('d-none');
                } else {
                    $('.consumo-detalhes-' + index).hide();
                    $('#consumo-placeholder-' + index).removeClass('d-none');
                    item.justificativa_consumo = '';
                    item.boletim_ocorrencia = '';
                    $('.justificativa-municao[data-index="' + index + '"]').val('');
                    $('.bo-municao[data-index="' + index + '"]').val('');
                }
                
                atualizarCampoOculto();
            });
            
            // Evento para extravios de muni├º├úo
            $('.extravios-municao').off('change keyup').on('change keyup', function() {
                const index = $(this).data('index');
                let extravios = parseInt($(this).val()) || 0;
                const item = materiaisSelecionados[index];
                
                if (extravios < 0) {
                    extravios = 0;
                    $(this).val(0);
                }
                
                if (item.disparos + extravios > item.quantidade) {
                    alert('A soma de disparos e extravios n├úo pode exceder a quantidade total de muni├º├Áes prestada de contas (' + item.quantidade + ').');
                    extravios = item.quantidade - item.disparos;
                    $(this).val(extravios);
                }
                
                item.extravios = extravios;
                
                const fisicos = item.quantidade - (item.disparos + item.extravios);
                $('#fisicos-' + index).text(fisicos);
                
                if (item.disparos > 0 || item.extravios > 0) {
                    $('.consumo-detalhes-' + index).show();
                    $('#consumo-placeholder-' + index).addClass('d-none');
                } else {
                    $('.consumo-detalhes-' + index).hide();
                    $('#consumo-placeholder-' + index).removeClass('d-none');
                    item.justificativa_consumo = '';
                    item.boletim_ocorrencia = '';
                    $('.justificativa-municao[data-index="' + index + '"]').val('');
                    $('.bo-municao[data-index="' + index + '"]').val('');
                }
                
                atualizarCampoOculto();
            });
            
            // Evento para justificativa de consumo
            $('.justificativa-municao').off('change keyup').on('change keyup', function() {
                const index = $(this).data('index');
                materiaisSelecionados[index].justificativa_consumo = $(this).val();
                atualizarCampoOculto();
            });
            
            // Evento para BO
            $('.bo-municao').off('change keyup').on('change keyup', function() {
                const index = $(this).data('index');
                materiaisSelecionados[index].boletim_ocorrencia = $(this).val();
                atualizarCampoOculto();
            });
            
            // Evento para atualizar o estado
            $('.estado-material').off('change').change(function() {
                const index = $(this).data('index');
                const novoEstado = $(this).val();
                materiaisSelecionados[index].estado = novoEstado;
                
                // Se o estado for RUIM ou PESSIMO, marca automaticamente para manuten├º├úo
                if (novoEstado === 'RUIM' || novoEstado === 'PESSIMO') {
                    $('#manutencao-' + index).prop('checked', true);
                    materiaisSelecionados[index].manutencao = true;
                }
                
                atualizarCampoOculto();
            });
            
            // Evento para atualizar a op├º├úo de manuten├º├úo
            $('.manutencao-material').off('change').change(function() {
                const index = $(this).data('index');
                const manutencao = $(this).prop('checked');
                materiaisSelecionados[index].manutencao = manutencao;
                atualizarCampoOculto();
            });
            
            // Evento para remover material
            $('.remover-material').off('click').click(function() {
                const index = $(this).data('index');
                removerMaterialDaDevolucao(index);
            });
        }
        
        // Fun├º├úo para carregar materiais pendentes
        function carregarMateriaisPendentes() {
            const policialId = $('#filtro-policial').val();
            const materialId = $('#filtro-material').val();
            
            // Mostra indicador de carregamento
            $('#materiais-pendentes-lista').html('<div class="text-center"><div class="spinner-border text-primary" role="status"><span class="visually-hidden">Carregando...</span></div></div>');
            
            $.ajax({
                url: '/movimentacoes/api/movimentacoes/retiradas-pendentes/',
                type: 'GET',
                dataType: 'json',
                data: {
                    policial_id: policialId,
                    material_id: materialId
                },
                success: function(data) {
                    if (Array.isArray(data)) {
                        materiaisPendentes = data;
                        if (data.length === 0) {
                            $('#materiais-pendentes-lista').html('<div class="alert alert-warning">N├úo h├í materiais pendentes de devolu├º├úo com os filtros aplicados.</div>');
                        } else {
                            exibirMateriaisPendentes(data);
                        }
                    } else {
                        console.error('Dados de retiradas pendentes n├úo s├úo um array:', data);
                        $('#materiais-pendentes-lista').html('<div class="alert alert-danger">Erro ao processar dados dos materiais pendentes.</div>');
                    }
                },
                error: function(xhr, status, error) {
                    console.error('Erro ao carregar materiais pendentes:', error);
                    $('#materiais-pendentes-lista').html('<div class="alert alert-danger">Erro ao carregar materiais pendentes.</div>');
                }
            });
        }
        
        // Fun├º├úo para exibir a lista de materiais pendentes
        function exibirMateriaisPendentes(retiradas) {
            if (!retiradas || retiradas.length === 0) {
                $('#materiais-pendentes-lista').html('<div class="alert alert-warning">N├úo h├í materiais pendentes de devolu├º├úo.</div>');
                return;
            }
            
            let html = '<div class="table-responsive"><table class="table table-bordered table-hover"><thead class="table-light"><tr>';
            html += '<th><input type="checkbox" id="select-all" class="form-check-input"></th>';
            html += '<th>Identifica├º├úo</th><th>Policial</th><th>Data Retirada</th><th>Finalidade</th><th>Qtd. Pendente</th>';
            html += '</tr></thead><tbody>';
            
            retiradas.forEach(function(item, index) {
                const isSelected = materiaisSelecionados.some(selected => 
                    selected.retirada_id === item.id && selected.material_id === item.material.id
                );
                
                let detailText = item.material.nome || '';
                if (item.lote) {
                    detailText += ` <span class="badge bg-secondary">Lote: ${item.lote.numero_lote}</span>`;
                }
                
                html += '<tr>';
                html += '<td><input type="checkbox" class="form-check-input material-checkbox" data-index="' + index + '"' + (isSelected ? ' checked' : '') + '></td>';
                html += '<td>' + detailText + (item.material.numero ? ' (' + item.material.numero + ')' : '') + '</td>';
                html += '<td>' + item.policial.re + ' - ' + item.policial.nome + '</td>';
                html += '<td>' + item.data_hora + '</td>';
                html += '<td>' + (item.finalidade || '-') + '</td>';
                html += '<td>' + item.quantidade_pendente + '</td>';
                html += '</tr>';
            });
            
            html += '</tbody></table></div>';
            
            $('#materiais-pendentes-lista').html(html);
            
            // Adiciona eventos para os checkboxes
            adicionarEventosCheckboxes();
        }
        
        // Fun├º├úo para adicionar eventos aos checkboxes
        function adicionarEventosCheckboxes() {
            // Evento para checkbox "selecionar todos"
            $('#select-all').change(function() {
                const isChecked = $(this).prop('checked');
                $('.material-checkbox').prop('checked', isChecked);
                
                if (isChecked) {
                    // Adiciona todos os materiais n├úo selecionados
                    materiaisPendentes.forEach(function(item, index) {
                        const isSelected = materiaisSelecionados.some(selected => 
                            selected.retirada_id === item.id && selected.material_id === item.material.id
                        );
                        
                        if (!isSelected) {
                            adicionarMaterialParaDevolucao(item);
                        }
                    });
                } else {
                    // Remove todos os materiais selecionados
                    materiaisSelecionados = [];
                    atualizarCampoOculto();
                    atualizarListaMateriais();
                }
            });
            
            // Evento para checkboxes individuais
            $('.material-checkbox').change(function() {
                const index = $(this).data('index');
                const isChecked = $(this).prop('checked');
                const item = materiaisPendentes[index];
                
                if (isChecked) {
                    // Adiciona o material
                    adicionarMaterialParaDevolucao(item);
                } else {
                    // Remove o material
                    const materialIndex = materiaisSelecionados.findIndex(selected => 
                        selected.retirada_id === item.id && selected.material_id === item.material.id
                    );
                    
                    if (materialIndex !== -1) {
                        removerMaterialDaDevolucao(materialIndex);
                    }
                }
                
                // Atualiza o estado do checkbox "selecionar todos"
                const totalCheckboxes = $('.material-checkbox').length;
                const checkedCheckboxes = $('.material-checkbox:checked').length;
                $('#select-all').prop('checked', totalCheckboxes === checkedCheckboxes);
            });
        }
        
        // Fun├º├úo para carregar filtros
        function carregarFiltros() {
            // Carrega policiais para o filtro
            $.ajax({
                url: '/policiais/api/policiais/',

                dataType: 'json',

                            option.text(policial.re + ' - ' + policial.nome);
                            select.append(option);
                        });
                    }
                }
            });
            
            // Carrega materiais para o filtro
            $.ajax({
                url: '/materiais/api/materiais/',
                type: 'GET',
                dataType: 'json',
                success: function(data) {
                    const select = $('#filtro-material');
                    select.empty();
                    select.append('<option value="">Todos os materiais</option>');
                    
                    if (Array.isArray(data)) {
                        data.forEach(function(material) {
                            const option = $('<option></option>');
                            option.val(material.id);
                            option.text(material.tipo_display + ' - ' + material.numero);
                            select.append(option);
                        });
                    }
                }
            });
        }
        
        // Eventos dos filtros
        $('#filtro-policial, #filtro-material').change(function() {
            carregarMateriaisPendentes();
        });
        
        $('#btn-limpar-filtros').click(function() {
            $('#filtro-policial').val('');
            $('#filtro-material').val('');

        });
        
        // Eventos dos bot├Áes de a├º├úo
        $('#btn-selecionar-todos').click(function() {
            $('#select-all').prop('checked', true).trigger('change');
        });
        
        $('#btn-limpar-selecao').click(function() {
            $('#select-all').prop('checked', false).trigger('change');
        });
        
        $('#btn-remover-todos').click(function() {
            if (confirm('Tem certeza que deseja remover todos os materiais selecionados?')) {
                materiaisSelecionados = [];
                atualizarCampoOculto();
                atualizarListaMateriais();
                $('.material-checkbox').prop('checked', false);
                $('#select-all').prop('checked', false);
            }
        });
        
        // Configura o Select2 com Ajax para buscar policiais
        $('#id_policial, #filtro-policial').select2({
            theme: 'bootstrap-5',
            placeholder: 'Digite RE ou nome do policial...',
            allowClear: true,
            minimumInputLength: 2,
            ajax: {

                delay: 250,

                processResults: function (data) {

                    };
                },
                cache: true
            }

        // Eventos de mudan├ºa para o policial principal
        $('#id_policial').on('select2:select', function(e) {

        });
        
        $('#id_policial').on('select2:clear', function(e) {
            carregarInfoPolicial(null);
        });

        // Evento de mudan├ºa para o filtro
        $('#filtro-policial').on('change', function() {
            carregarMateriaisPendentes();
        });
        
        // Inicializa├º├úo
        carregarFiltros();
        carregarMateriaisPendentes();
        atualizarListaMateriais();
        
        // Valida├º├úo do formul├írio
        $('form').submit(function(e) {
            if (materiaisSelecionados.length === 0) {
                e.preventDefault();
                alert('Selecione pelo menos um material para devolu├º├úo.');
                return false;
            }
            
            // Valida se todo consumo de muni├º├úo tem justificativa preenchida
            let erroJustificativa = false;
            materiaisSelecionados.forEach(function(item) {
                if (item.raw_tipo === 'MUNICAO') {
                    if ((item.disparos > 0 || item.extravios > 0) && !item.justificativa_consumo.trim()) {
                        erroJustificativa = true;
                    }
                }
            });
            
            if (erroJustificativa) {
                e.preventDefault();
                alert('A justificativa ├® obrigat├│ria sempre que houver muni├º├Áes disparadas ou extraviadas.');
                return false;
            }
            
            return true;
        });
    });
